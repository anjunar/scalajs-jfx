The same reactive counter in both APIs

Scala 3Run example →
import jfx.core.dsl.EventDsl.onClick
import jfx.core.layout.Button.button
import jfx.core.layout.TextComponent.text
import jfx.core.layout.VBox.vbox
import jfx.core.state.Property

val count = Property(0)

vbox {
  text(count.map(n => s"Count: $n")) {}
  button("Increment") {
    onClick(_ => count.set(count.get + 1))
  }
}
TypeScriptRun example →
import { button, onClick, property, text, vbox } from "@anjunar/jfx-core";

const count = property(0);

vbox(() => {
  text(count.map(n => "Count: " + n));
  button("Increment", {}, () =>
    onClick(() => count.set(count.get + 1))
  );
});
